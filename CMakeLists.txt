cmake_minimum_required(VERSION 3.16)
project(diar_gpu_engine VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(diar_core STATIC src/diar.cpp src/aosc.cpp src/birth_gate.cpp src/fe.cpp src/nn.cpp src/layers.cpp src/mha.cpp src/conv.cpp src/conformer.cpp src/subsampling.cpp src/posenc.cpp src/gguf.cpp src/sortformer.cpp)
target_include_directories(diar_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(diar_core PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(diar_core PRIVATE /W4)
else()
  target_compile_options(diar_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
